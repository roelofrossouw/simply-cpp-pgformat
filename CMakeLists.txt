cmake_minimum_required(VERSION 3.22)
set(SC_MODULE pgformat)
project(simply-cpp-${SC_MODULE} HOMEPAGE_URL "https://github.com/roelofrossouw/simply-cpp-pgformat" LANGUAGES CXX)
include(cmake/sc_bootstrap.cmake)

option(SC_PGFORMAT_BUILD_TESTS "Build simply-cpp-pgformat tests" ${PROJECT_IS_TOP_LEVEL})

get_sc_version()
message(STATUS "SimplyCPP ${SC_MODULE} Version: ${SC_VERSION}")

# sc::params (and its man pages) for pgformat.
find_or_install_package(sc-core simply-cpp-dev simply-cpp 2.1.0)
# PCRE2's 32-bit library: pgFormatter's Perl regular expressions, on text as Perl sees it (a
# character an element).
config_or_install_package(libpcre2-32 libpcre2-dev pcre2)
if (SC_PGFORMAT_PCRE2_LIBRARY AND NOT EXISTS "${SC_PGFORMAT_PCRE2_LIBRARY}")
    message(STATUS "pcre2-32: the cached library location is gone, looking again")
    unset(SC_PGFORMAT_PCRE2_LIBRARY CACHE)
endif ()
find_library(SC_PGFORMAT_PCRE2_LIBRARY NAMES pcre2-32 HINTS ${libpcre2-32_LIBRARY_DIRS} REQUIRED)
# This path goes into the exported targets, so not Homebrew's versioned Cellar folder (removed on
# every upgrade) but its opt link, which follows.
if (SC_PGFORMAT_PCRE2_LIBRARY MATCHES "^(.*)/Cellar/([^/]+)/[^/]+/(.*)$" AND EXISTS "${CMAKE_MATCH_1}/opt/${CMAKE_MATCH_2}/${CMAKE_MATCH_3}")
    set(SC_PGFORMAT_PCRE2_LIBRARY "${CMAKE_MATCH_1}/opt/${CMAKE_MATCH_2}/${CMAKE_MATCH_3}" CACHE FILEPATH "" FORCE)
endif ()

set(SOURCE_OBJECTS)
set(SOURCE_LIBRARIES)
set(SOURCE_LINK_LIBRARIES)

# The port of pgFormatter's Beautify.pm: the Perl helpers (perl.cpp), the dictionaries (dicts.cpp,
# generated by tools/dump_dicts.pl), Beautify's functions (beautify.cpp) and its token loop
# (beautify_run.cpp). PCRE2 stays out of the installed header.
add_sc_object(sc_pgformat
        SOURCES src/perl.h src/perl.cpp src/dicts.h src/dicts.cpp src/beautify.h src/beautify.cpp src/beautify_run.cpp
        INCLUDE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}/src
        LINK_LIBRARIES PkgConfig::libpcre2-32
        PUBLIC_LINK_LIBRARIES ${SC_PGFORMAT_PCRE2_LIBRARY})

add_sc_libraries(DESCRIPTION "simply-cpp PostgreSQL SQL and PL/pgSQL formatting, as pgFormatter does it")
target_compile_features(sc-pgformat PUBLIC cxx_std_20)
target_compile_features(sc-pgformat-shared PUBLIC cxx_std_20)

install_sc_module(PATH_VARS CMAKE_INSTALL_INCLUDEDIR)
install(FILES cmake/SimplyCppFunctions.cmake cmake/sc_bootstrap.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/sc-pgformat
        COMPONENT development)

if (SC_PGFORMAT_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests sc-pgformat-test)
endif ()

add_sc_example(sc-pgformat-demo)

# pgformat, the program: pg_format's options, by the same letters; named pgformat so it can be
# installed beside pgFormatter's pg_format.
add_executable(pgformat apps/pgformat.cpp)
target_link_libraries(pgformat PRIVATE sc::sc-pgformat-shared sc::sc-core)
target_compile_definitions(pgformat PRIVATE SC_PGFORMAT_VERSION="${SC_VERSION}")
sc_program_info(pgformat)
install(TARGETS pgformat RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT runtime)
sc_man_page(pgformat EXTRA ${CMAKE_CURRENT_SOURCE_DIR}/man/pgformat.md COMPONENT runtime)

package_sc_module(DEPENDS "simply-cpp (>= 2.1.0~)" DEV_DEPENDS "simply-cpp-dev (>= 2.1.0~)" libpcre2-dev)
